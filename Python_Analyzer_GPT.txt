
import csv

class SalesAnalyzer:
    """
    A class to analyze sales data from a CSV file.
    
    Provides methods to load sales data and calculate metrics such as total sales
    and the top-selling product by revenue.
    """

    def __init__(self, filename):
        """
        Initialize the SalesAnalyzer with a CSV filename.
        
        Args:
            filename (str): Path to the CSV file containing sales data.
                           Expected columns: 'product_name', 'price', 'quantity'
        """
        self.filename = filename
        self.data = []

    def load_data(self):
        """
        Load sales data from the CSV file.
        
        Reads the CSV file and stores all rows as dictionaries.
        Returns True on success, False on failure.
        
        Returns:
            bool: True if data was loaded successfully, False otherwise.
        """
        try:
            with open(self.filename, mode='r', encoding='utf-8', newline='') as file:
                csv_reader = csv.DictReader(file)

                for line_num, row in enumerate(csv_reader, start=2):
                    try:
                        # Validate required columns and numeric values
                        product = row['product_name']
                        price = float(row['price'])
                        quantity = int(row['quantity'])
                    except KeyError as exc:
                        print(f"Error: Missing required column {exc} in CSV header.")
                        return False
                    except ValueError as exc:
                        print(
                            f"Error: Invalid numeric value on line {line_num}: {exc}. "
                            f"Expected numeric values for price and quantity."
                        )
                        return False

                    self.data.append(row)

        except FileNotFoundError:
            print(f"Error: File not found: {self.filename}")
            return False
        except OSError as exc:
            print(f"Error reading file {self.filename}: {exc}")
            return False

        return True

    def calculate_total_sales(self):
        """
        Calculate the total sales revenue from all products.
        
        Sums up the revenue (price * quantity) for all items in the loaded data.
        Returns the total as a float, or None if data has not been loaded.
        
        Returns:
            float or None: Total sales revenue, or None if data is not loaded.
        """
        if not self.data:
            print("Error: No data loaded. Call load_data() first.")
            return None

        total = 0.0
        for row in self.data:
            try:
                price = float(row['price'])
                quantity = int(row['quantity'])
                total += price * quantity
            except (KeyError, ValueError) as exc:
                print(f"Error processing row: {exc}")
                return None

        return total

    def find_top_product(self):
        """
        Identify the top-selling product by total revenue.
        
        Aggregates revenue (price * quantity) for each product and returns
        the product with the highest total revenue.
        Returns a tuple (product_name, revenue) or (None, None) on error.
        
        Returns:
            tuple: (product_name, revenue) where product_name is a string
                   and revenue is a float, or (None, None) if data is not loaded.
        """
        if not self.data:
            print("Error: No data loaded. Call load_data() first.")
            return None, None

        revenue_by_product = {}

        for row in self.data:
            try:
                product = row['product_name']
                price = float(row['price'])
                quantity = int(row['quantity'])
            except (KeyError, ValueError) as exc:
                print(f"Error processing row: {exc}")
                return None, None

            revenue_by_product[product] = revenue_by_product.get(product, 0.0) + price * quantity

        if not revenue_by_product:
            print(f"No sales data found in {self.filename}.")
            return None, None

        top_product = max(revenue_by_product.items(), key=lambda x: x[1])
        return top_product


if __name__ == "__main__":
    sales_data_file = 'sales_data.csv'
    analyzer = SalesAnalyzer(sales_data_file)

    if analyzer.load_data():
        total_sales = analyzer.calculate_total_sales()
        if total_sales is not None:
            print(f"Total sales from {sales_data_file}: ${total_sales:.2f}")
        else:
            print("Total sales could not be calculated due to earlier errors.")

        top_product, top_revenue = analyzer.find_top_product()
        if top_product is not None:
            print(f"Top-selling product by revenue: {top_product} (${top_revenue:.2f})")
        else:
            print("Top-selling product could not be determined due to earlier errors.")
    else:
        print("Failed to load sales data.")
